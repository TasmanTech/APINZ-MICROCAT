def guarded(fn):
    """A failed step must SAY it failed, on the screen, in words."""
    def wrap(*a, **k):
        try:
            return fn(*a, **k)
        except Exception as e:
            traceback.print_exc()
            return jsonify({"error": f"{type(e).__name__}: {str(e)[:300]}"}), 200
    wrap.__name__ = fn.__name__
    return wrap

        return Response(f.read(), mimetype="text/html")


def _open_refs():
    try:
        with open(os.path.join(HERE, "open_now.json"), encoding="utf-8") as f:
            return set(json.load(f).get("refs") or [])
    except Exception:
        return None
